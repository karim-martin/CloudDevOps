Monitoring on AWS

AWS

CloudWatch Container Insights, alarms, and dashboards for EKS.

Enable Container Insights

# Enable Container Insights for EKS
aws eks create-addon \
  --cluster-name myapp-cluster \
  --addon-name amazon-cloudwatch-observability \
  --service-account-role-arn arn:aws:iam::123:role/CloudWatchAgentRole \
  --region us-east-1

Create CloudWatch Dashboard (Terraform)

# Important metrics to track:
# - Pod CPU/Memory utilization
# - HTTP 5xx error rate from ALB
# - RDS connections and read/write IOPS
# - EKS node disk pressure

resource "aws_cloudwatch_dashboard" "main" {
  dashboard_name = "myapp-production"
  dashboard_body = jsonencode({
    widgets = [
      {
        type = "metric"
        properties = {
          metrics = [
            ["AWS/ApplicationELB", "HTTPCode_Target_5XX_Count", "LoadBalancer", aws_lb.main.arn_suffix]
          ]
          period = 60
          title  = "5xx Errors"
        }
      },
      {
        type = "metric"
        properties = {
          metrics = [
            ["ContainerInsights", "pod_cpu_utilization", "Namespace", "production", "ClusterName", "myapp-cluster"]
          ]
          title = "Pod CPU Utilization"
        }
      }
    ]
  })
}
