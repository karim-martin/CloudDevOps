S3 Storage & ECR

S3 for file uploads and static assets; ECR for Docker images.

S3 Bucket for Uploads

# s3.tf
resource "aws_s3_bucket" "uploads" {
  bucket = "myapp-uploads-${data.aws_caller_identity.current.account_id}"
  tags   = { Environment = "production", Project = "myapp" }
}

resource "aws_s3_bucket_public_access_block" "uploads" {
  bucket                  = aws_s3_bucket.uploads.id
  block_public_acls       = true
  block_public_policy     = true
  ignore_public_acls      = true
  restrict_public_buckets = true
}

resource "aws_s3_bucket_server_side_encryption_configuration" "uploads" {
  bucket = aws_s3_bucket.uploads.id
  rule {
    apply_server_side_encryption_by_default {
      sse_algorithm = "AES256"
    }
  }
}

resource "aws_s3_bucket_cors_configuration" "uploads" {
  bucket = aws_s3_bucket.uploads.id
  cors_rule {
    allowed_headers = ["*"]
    allowed_methods = ["GET", "POST", "PUT"]
    allowed_origins = ["https://myapp.com"]
    max_age_seconds = 3600
  }
}

data "aws_caller_identity" "current" {}

ECR Repositories

# ecr.tf
resource "aws_ecr_repository" "frontend" {
  name                 = "myapp/frontend"
  image_tag_mutability = "IMMUTABLE"

  image_scanning_configuration {
    scan_on_push = true
  }

  encryption_configuration {
    encryption_type = "AES256"
  }
}

resource "aws_ecr_repository" "backend" {
  name                 = "myapp/backend"
  image_tag_mutability = "IMMUTABLE"

  image_scanning_configuration {
    scan_on_push = true
  }
}

# Lifecycle policy — keep only the last 20 images
resource "aws_ecr_lifecycle_policy" "frontend" {
  repository = aws_ecr_repository.frontend.name
  policy = jsonencode({
    rules = [{
      rulePriority = 1
      description  = "Keep last 20 images"
      selection    = { tagStatus = "any", countType = "imageCountMoreThan", countNumber = 20 }
      action       = { type = "expire" }
    }]
  })
}

Push Images to ECR

# Authenticate Docker
aws ecr get-login-password --region us-east-1 | \
  docker login --username AWS --password-stdin \
  $(aws sts get-caller-identity --query Account --output text).dkr.ecr.us-east-1.amazonaws.com

# Tag and push
ACCT=$(aws sts get-caller-identity --query Account --output text)
docker tag myapp-frontend:v1 $ACCT.dkr.ecr.us-east-1.amazonaws.com/myapp/frontend:v1
docker push                  $ACCT.dkr.ecr.us-east-1.amazonaws.com/myapp/frontend:v1
Immutable Tags Prevent Mistakes

image_tag_mutability = "IMMUTABLE" stops anyone from overwriting an existing tag. This makes rollbacks safe — v1.2.3 always points to the same image.
