RDS for SQL Server

Managed SQL Server on RDS for the .NET 8 backend — private, Multi-AZ, with automatic backups.

Create RDS with Terraform

# rds.tf
resource "aws_db_subnet_group" "main" {
  name       = "myapp-db-subnets"
  subnet_ids = module.vpc.database_subnets
  tags       = { Name = "myapp-db-subnets" }
}

resource "aws_security_group" "rds" {
  name        = "myapp-rds-sg"
  description = "Allow SQL Server from EKS nodes only"
  vpc_id      = module.vpc.vpc_id

  ingress {
    from_port       = 1433
    to_port         = 1433
    protocol        = "tcp"
    security_groups = [module.eks.node_security_group_id]
  }
}

resource "aws_db_instance" "main" {
  identifier             = "myapp-sqlserver"
  engine                 = "sqlserver-ex"
  engine_version         = "15.00.4335.1.v1"
  instance_class         = "db.t3.small"
  allocated_storage      = 20
  storage_encrypted      = true

  username               = "sqladmin"
  password               = var.db_password
  db_subnet_group_name   = aws_db_subnet_group.main.name
  vpc_security_group_ids = [aws_security_group.rds.id]

  multi_az                = true
  backup_retention_period = 7
  skip_final_snapshot     = false
  final_snapshot_identifier = "myapp-sqlserver-final"

  deletion_protection = true
  tags = { Environment = "production", Project = "myapp" }
}

output "rds_endpoint" { value = aws_db_instance.main.address }

.NET 8 Connection String

// appsettings.json
{
  "ConnectionStrings": {
    "DefaultConnection": "Server=myapp-sqlserver.xxxxx.us-east-1.rds.amazonaws.com,1433;Database=myapp;User Id=sqladmin;Password=${DB_PASSWORD};TrustServerCertificate=True;"
  }
}

// Program.cs — Entity Framework Core
builder.Services.AddDbContext<AppDbContext>(options =>
    options.UseSqlServer(builder.Configuration.GetConnectionString("DefaultConnection")));
Store the Password in Secrets Manager

Don't put the DB password in appsettings.json or a plain Kubernetes Secret. Store it in AWS Secrets Manager and retrieve it at runtime via IRSA — see the IRSA section.
