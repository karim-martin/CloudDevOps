AWS Foundation

AWS

AWS account setup, IAM, and VPC networking — the base for your EKS deployment.

Before deploying to EKS, you need an AWS account with IAM configured and a VPC with public/private subnets set up.

IAM Setup

# Create an admin IAM user for deployment (avoid root)
aws iam create-user --user-name eks-deployer

# Attach AdministratorAccess policy (tighten later)
aws iam attach-user-policy \
  --user-name eks-deployer \
  --policy-arn arn:aws:iam::aws:policy/AdministratorAccess

# Create access keys
aws iam create-access-key --user-name eks-deployer

VPC Setup with Terraform

# main.tf
terraform {
  required_providers {
    aws = { source = "hashicorp/aws", version = "~> 5.0" }
  }
}

provider "aws" {
  region = "us-east-1"
}

module "vpc" {
  source  = "terraform-aws-modules/vpc/aws"
  version = "5.0"

  name = "myapp-vpc"
  cidr = "10.0.0.0/16"

  azs             = ["us-east-1a", "us-east-1b", "us-east-1c"]
  public_subnets  = ["10.0.1.0/24", "10.0.2.0/24", "10.0.3.0/24"]
  private_subnets = ["10.0.11.0/24", "10.0.12.0/24", "10.0.13.0/24"]
  database_subnets = ["10.0.21.0/24", "10.0.22.0/24", "10.0.23.0/24"]

  enable_nat_gateway     = true
  one_nat_gateway_per_az = true

  # Required for EKS to discover subnets
  public_subnet_tags  = { "kubernetes.io/role/elb" = "1" }
  private_subnet_tags = { "kubernetes.io/role/internal-elb" = "1" }

  tags = { Environment = "production", Project = "myapp" }
}

# Output VPC and subnet IDs for use in other modules
output "vpc_id" { value = module.vpc.vpc_id }
output "private_subnet_ids" { value = module.vpc.private_subnets }
output "public_subnet_ids" { value = module.vpc.public_subnets }
output "database_subnet_ids" { value = module.vpc.database_subnets }
terraform init
terraform plan
terraform apply  # Takes ~5 minutes
