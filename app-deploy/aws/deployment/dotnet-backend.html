.NET 8 Backend on EKS

AWS

Kubernetes Deployment for the .NET 8 API with IRSA and Secrets Manager.

Service Account with IRSA

# Create IAM role for the backend pod
eksctl create iamserviceaccount \
  --cluster myapp-cluster \
  --namespace production \
  --name backend-sa \
  --attach-policy-arn arn:aws:iam::123:policy/MyAppBackendPolicy \
  --approve

Backend Deployment

# k8s/backend/deployment.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: backend
  namespace: production
spec:
  replicas: 3
  selector:
    matchLabels:
      app: backend
  template:
    metadata:
      labels:
        app: backend
    spec:
      serviceAccountName: backend-sa  # IRSA — pod gets AWS permissions
      containers:
      - name: backend
        image: 123456789.dkr.ecr.us-east-1.amazonaws.com/myapp/backend:latest
        ports:
        - containerPort: 8080
        env:
        - name: AWS_REGION
          value: us-east-1
        - name: DB_SECRET_NAME
          value: myapp/prod/database
        # .NET reads DB credentials from Secrets Manager at startup (via AWS SDK + IRSA)
        resources:
          requests: {memory: "256Mi", cpu: "250m"}
          limits:   {memory: "512Mi", cpu: "500m"}
        livenessProbe:
          httpGet: {path: /health, port: 8080}
          initialDelaySeconds: 30
          periodSeconds: 30
---
apiVersion: v1
kind: Service
metadata:
  name: backend-svc
  namespace: production
spec:
  selector:
    app: backend
  ports:
  - port: 8080
    targetPort: 8080

Read Secrets in .NET 8

// Program.cs
// Uses AWS SDK — IRSA provides credentials automatically
builder.Services.AddAWSService<IAmazonSecretsManager>();
builder.Services.AddSingleton<IDbConfig>(sp => {
    var sm = sp.GetRequiredService<IAmazonSecretsManager>();
    var response = sm.GetSecretValueAsync(new GetSecretValueRequest {
        SecretId = Environment.GetEnvironmentVariable("DB_SECRET_NAME")
    }).GetAwaiter().GetResult();
    return JsonSerializer.Deserialize<DbConfig>(response.SecretString)!;
});
