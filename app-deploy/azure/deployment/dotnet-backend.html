Deploy .NET 8 Backend

AKS deployment with Workload Identity so pods access Key Vault without passwords.

Kubernetes Deployment

# backend-deployment.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: backend
  namespace: myapp
spec:
  replicas: 2
  selector:
    matchLabels:
      app: backend
  template:
    metadata:
      labels:
        app: backend
        azure.workload.identity/use: "true"
    spec:
      serviceAccountName: backend-sa   # linked to Managed Identity
      containers:
      - name: backend
        image: myappacr.azurecr.io/backend:latest
        ports:
        - containerPort: 8080
        env:
        - name: ASPNETCORE_ENVIRONMENT
          value: "Production"
        - name: KeyVaultUri
          value: "https://myapp-kv.vault.azure.net/"
        resources:
          requests:
            memory: "256Mi"
            cpu: "250m"
          limits:
            memory: "512Mi"
            cpu: "500m"
---
apiVersion: v1
kind: Service
metadata:
  name: backend-service
  namespace: myapp
spec:
  selector:
    app: backend
  ports:
  - port: 8080
    targetPort: 8080
  type: ClusterIP

.NET 8 — Read Secrets from Key Vault

// Program.cs
using Azure.Identity;

var builder = WebApplication.CreateBuilder(args);

// Add Key Vault — uses Workload Identity automatically
var keyVaultUri = new Uri(builder.Configuration["KeyVaultUri"]!);
builder.Configuration.AddAzureKeyVault(keyVaultUri, new DefaultAzureCredential());

builder.Services.AddDbContext<AppDbContext>(options =>
    options.UseSqlServer(builder.Configuration["ConnectionStrings--DefaultConnection"]));

builder.Services.AddControllers();
var app = builder.Build();
app.MapControllers();
app.Run();

Apply the Deployment

kubectl apply -f backend-deployment.yaml -n myapp
kubectl rollout status deployment/backend -n myapp
