Azure SQL Database

Managed SQL Server for your .NET 8 application — with private endpoint and automatic backups.

Create Azure SQL with Terraform

# sql.tf
resource "azurerm_mssql_server" "main" {
  name                         = "myapp-sql-server"
  resource_group_name          = azurerm_resource_group.main.name
  location                     = azurerm_resource_group.main.location
  version                      = "12.0"
  administrator_login          = "sqladmin"
  administrator_login_password = var.sql_admin_password

  azuread_administrator {
    login_username = "AzureAD Admin"
    object_id      = data.azurerm_client_config.current.object_id
  }
}

resource "azurerm_mssql_database" "main" {
  name           = "myapp-db"
  server_id      = azurerm_mssql_server.main.id
  collation      = "SQL_Latin1_General_CP1_CI_AS"
  sku_name       = "S1"
  max_size_gb    = 10
  zone_redundant = false

  short_term_retention_policy {
    retention_days = 7
  }
}

# Firewall rule — allow Azure services
resource "azurerm_mssql_firewall_rule" "azure_services" {
  name             = "AllowAzureServices"
  server_id        = azurerm_mssql_server.main.id
  start_ip_address = "0.0.0.0"
  end_ip_address   = "0.0.0.0"
}

# VNet rule — restrict to AKS subnet
resource "azurerm_mssql_virtual_network_rule" "aks" {
  name      = "aks-subnet"
  server_id = azurerm_mssql_server.main.id
  subnet_id = azurerm_subnet.aks_nodes.id
}

.NET 8 Connection String

// appsettings.json — connection string stored in Key Vault
{
  "ConnectionStrings": {
    "DefaultConnection": "Server=myapp-sql-server.database.windows.net;Database=myapp-db;Authentication=Active Directory Default;"
  }
}

// Program.cs — using Entity Framework Core
builder.Services.AddDbContext<AppDbContext>(options =>
    options.UseSqlServer(builder.Configuration.GetConnectionString("DefaultConnection")));
Use Active Directory Auth — No Passwords

The connection string above uses Authentication=Active Directory Default — your AKS pod's Managed Identity authenticates automatically. No password needed, no credentials to rotate.
