AWS CodeBuild

AWS

Managed build service — compile, test, and package applications.

CodeBuild = A Build Server You Never Manage

CodeBuild provisions a fresh container environment, runs your build commands, and tears it down. You pay only for build minutes used. No Jenkins server to maintain, no build agents to scale. Each build gets an isolated environment — builds can't affect each other. It integrates natively with ECR, S3, and Secrets Manager.

CodeBuild Project

resource "aws_codebuild_project" "build" {
  name          = "myapp-build"
  service_role  = aws_iam_role.codebuild.arn
  build_timeout = 20  # minutes

  environment {
    compute_type    = "BUILD_GENERAL1_SMALL"
    image           = "aws/codebuild/standard:7.0"
    type            = "LINUX_CONTAINER"
    privileged_mode = true  # needed for Docker builds

    environment_variable {
      name  = "ECR_URI"
      value = aws_ecr_repository.app.repository_url
    }
  }

  source {
    type      = "GITHUB"
    location  = "https://github.com/myorg/myapp"
    buildspec = "buildspec.yml"
  }

  artifacts { type = "NO_ARTIFACTS" }

  logs_config {
    cloudwatch_logs { group_name = "/codebuild/myapp" }
  }
}

.NET 8 Build Example

# buildspec.yml for .NET 8 API
version: 0.2
phases:
  install:
    runtime-versions:
      dotnet: 8.0
  pre_build:
    commands:
      - dotnet restore
      - aws ecr get-login-password | docker login --username AWS --password-stdin $ECR_URI
  build:
    commands:
      - dotnet test --no-restore --logger "trx;LogFileName=results.xml"
      - docker build -f Dockerfile -t $ECR_URI/api:$CODEBUILD_RESOLVED_SOURCE_VERSION .
      - docker push $ECR_URI/api:$CODEBUILD_RESOLVED_SOURCE_VERSION
reports:
  test-results:
    files: ["**/results.xml"]
    file-format: VisualStudioTRX
