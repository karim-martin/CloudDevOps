Cloud-native CI/CD for any language, any platform, any cloud.

Azure Pipelines Automates Build and Deploy

Azure Pipelines runs your build and deployment steps automatically on every code push. It's defined in YAML (azure-pipelines.yml). Microsoft-hosted agents (free tier: 1800 min/month) provide pre-installed environments for .NET, Node.js, Java, Python, Docker, and kubectl. Self-hosted agents run on your own machines for more control.

Complete .NET + AKS Pipeline

# azure-pipelines.yml
trigger:
  branches:
    include: [main]

variables:
  ACR_NAME: myappcr
  AKS_CLUSTER: myapp-cluster
  AKS_RG: myapp-rg
  IMAGE_TAG: $(Build.BuildId)

stages:
- stage: Build
  jobs:
  - job: BuildAndTest
    pool:
      vmImage: ubuntu-latest
    steps:
    - task: DotNetCoreCLI@2
      displayName: Restore
      inputs: {command: restore, projects: '**/*.csproj'}

    - task: DotNetCoreCLI@2
      displayName: Build
      inputs: {command: build, arguments: '--no-restore --configuration Release'}

    - task: DotNetCoreCLI@2
      displayName: Test
      inputs: {command: test, arguments: '--no-build --configuration Release --collect:"Code Coverage"'}

    - task: Docker@2
      displayName: Build and push image
      inputs:
        command: buildAndPush
        containerRegistry: myapp-acr-service-connection
        repository: api
        tags: $(IMAGE_TAG)

- stage: Deploy
  condition: succeeded()
  jobs:
  - deployment: DeployToAKS
    environment: production
    pool:
      vmImage: ubuntu-latest
    strategy:
      runOnce:
        deploy:
          steps:
          - task: KubernetesManifest@0
            inputs:
              action: deploy
              kubernetesServiceConnection: myapp-aks-connection
              manifests: k8s/deployment.yaml
              containers: $(ACR_NAME).azurecr.io/api:$(IMAGE_TAG)
